Key SQL Queries
1. Total emission by user (JOIN + GROUP BY)
-- Get monthly emission summary per user SELECT u.name, u.location, SUM(e.total_co2_kg) AS total_emission, AVG(e.total_co2_kg) AS avg_monthly, COUNT(a.activity_id) AS activity_count FROM user u INNER JOIN carbon_emission e ON u.user_id = e.user_id INNER JOIN carbon_activity a ON e.activity_id = a.activity_id WHERE MONTH(a.activity_date) = 12 GROUP BY u.user_id ORDER BY total_emission ASC;
2. Calculate & insert carbon emission
-- Carbon calculation formula -- Electricity: kWh × 0.716 = CO₂ kg -- Fuel: litres × 2.31 = CO₂ kg -- Transport: km × 0.21 = CO₂ kg INSERT INTO carbon_emission ( activity_id, user_id, electricity_co2, transport_co2, fuel_co2, waste_co2, total_co2_kg, emission_level ) VALUES ( 28, 1, 124 * 0.716, 840 * 0.21, 45 * 2.31, 18.5 * 0.994, 247.3, 'low' );
3. Recommendations with reduction potential
-- Fetch applicable recommendations SELECT r.title, r.description, r.co2_reduction_kg, r.category FROM recommendation r WHERE r.target_emission <= ( SELECT MAX(total_co2_kg) FROM carbon_emission WHERE user_id = 1 ) ORDER BY r.co2_reduction_kg DESC LIMIT 5;
4. Monthly report generation
-- Generate and store monthly report INSERT INTO report ( user_id, report_month, total_co2_kg, reduction_pct, sustainability_score ) SELECT e.user_id, DATE_FORMAT(NOW(), '%Y-%m-01'), SUM(e.total_co2_kg), (1 - SUM(e.total_co2_kg)/ LAG(SUM(e.total_co2_kg)) ) * 100, ROUND((1 - SUM(e.total_co2_kg)/412) * 100) FROM carbon_emission e WHERE e.user_id = 1 GROUP BY e.user_id;
